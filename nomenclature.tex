


% \mathcal (Calligraphic) Symbols
\nomenclature[M]{$\mathcal{N}$}{Total kg-moles per unit mass.}
\nomenclature[M]{$\mathcal{N}_j$}{Number of kg-moles per unit mass of species $j$.}
\nomenclature[M]{$\mathcal{L}$}{The Lagrangian}
\nomenclature[M]{$\mathcal{M}_j$}{Molecular weight of species $j$ in kg / kg-mole}

% Roman Symbols
\nomenclature[R]{$a_{ij}$}{Stoichiometric coefficients, e.g. number of atoms of element $i$ in species $j$. Weighted by element $i$'s molecular weight}
\nomenclature[R]{$\hat{R}$}{Universal gas constant ($\approx$ 8314 kJ/kg-K)}
\nomenclature[R]{$b_i$}{Sum of contributions from each species towards constraint for element $i$, i.e. $\sum_{j=1}^\text{NS} a_{i,j}\mathcal{N}_j$}
\nomenclature[R]{$p_\text{ref}$}{Reference pressure (typically 1 atm or 1 bar).}
\nomenclature[R]{$G$}{Gibbs free-energy}
\nomenclature[R]{$F$}{Helmholtz free-energy}
\nomenclature[R]{$T$}{Temperature (K)}
\nomenclature[R]{$q_j$}{Charge of species $j$ (0 for neutrals, 1 for ions, -1 for electrons)}
\nomenclature[R]{$V$}{Specific volume ($m^3/$kg)}
\nomenclature[R]{$\rho$}{Density ($kg/m^3)$}
\nomenclature[R]{$b_i^\circ$}{Specified mass fraction of element $i$}
\nomenclature[R]{$U_j^\circ$}{Standard state internal energy of species $j$ (J/kg)}
\nomenclature[R]{$H_j^\circ$}{Standard state enthalpy of species $j$ (J/kg)}.
\nomenclature[R]{$c_{v,j}^\circ$}{Standard state specific heat of species $j$}
\nomenclature[R]{$c_{p,j}^\circ$}{Standard state specific heat of species $j$}
\nomenclature[R]{$u_0'$}{User-specified internal energy (J/kg)}
\nomenclature[R]{$h_0'$}{User-specified enthalpy (J/kg)}
\nomenclature[R]{$u'$}{Sum of contributions from each species towards internal energy constraint, i.e. $\sum_{j}^\text{NS} \mathcal{N}_j U_j^\circ$}
\nomenclature[R]{$h'$}{Sum of contributions from each species towards enthalpy  constraint, i.e. $\sum_{j}^\text{NS} \mathcal{N}_j H_j^\circ$}

% Greek Symbols
\nomenclature[G]{$\lambda$}{Lagrange multiplier}
\nomenclature[G]{$\mu_j$}{Chemical potential of species $j$}
\nomenclature[G]{$\mu_j^\circ$}{Standard-state chemical potential of species $j$}
